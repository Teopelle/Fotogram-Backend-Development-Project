CREATE TABLE UTENTE (
	Mail			VARCHAR (100)	NOT NULL,
	Username		VARCHAR (30)	NOT NULL,
	Passw			VARCHAR (300)	NOT NULL,
	Immagine		TEXT,
	numFollow		INT		NOT NULL DEFAULT 0,
	numFollowers		INT		NOT NULL DEFAULT 0,
	dataNomina		TIMESTAMP,		
	èModeratore		BOOLEAN		NOT NULL DEFAULT FALSE,
	èAmministratore		BOOLEAN		NOT NULL DEFAULT FALSE,
	postModerati		INT		NOT NULL DEFAULT 0,
	èBannato		BOOLEAN		NOT NULL DEFAULT FALSE,
	PRIMARY KEY (mail, username, passw)
);

CREATE TABLE SESSIONE (
	IdSessione		SERIAL		PRIMARY KEY,
	DataLogin		TIMESTAMP	NOT NULL,
	MailUtente		VARCHAR (100)	NOT NULL,
	UsernameUtente		VARCHAR (30)	NOT NULL,
	PasswordUtente		VARCHAR (300)	NOT NULL,
	FOREIGN KEY (mailUtente, usernameUtente, passwordUtente)
		REFERENCES UTENTE (mail, username, passw)
		ON DELETE CASCADE
);


CREATE TABLE POST (
	Id			SERIAL		PRIMARY KEY,
	Username 		VARCHAR (30)	NOT NULL,
	NumLike			INT		NOT NULL DEFAULT 0,
	NumFlag			INT		NOT NULL DEFAULT 0,
	èModerato		BOOLEAN		NOT NULL DEFAULT FALSE,
	Data			TIMESTAMP	NOT NULL DEFAULT now(),
	eTesto			BOOLEAN		NOT NULL,
	eImmagine		BOOLEAN		NOT NULL,
	TextContent		TEXT,
	ImageUrl		TEXT,
	CONSTRAINT check_tipo CHECK (
		(eTesto = TRUE AND eImmagine = FALSE AND 				TextContent IS NOT NULL AND ImageUrl IS NULL) 			OR 	
		(eTesto = FALSE AND eImmagine = TRUE AND 				TextContent IS NULL AND ImageUrl IS NOT NULL)
	)
);

CREATE TABLE MODERA (
	MailU			VARCHAR (100)	NOT NULL,
	UsernameU		VARCHAR (30)	NOT NULL,
	PasswU			VARCHAR (300)	NOT NULL,
	IdPost			INT		NOT NULL,
	DataModerazione		TIMESTAMP	NOT NULL,
	PRIMARY KEY (mailU, usernameU, passwU, idPost),
	FOREIGN KEY (mailU, usernameU, passwU)
		REFERENCES UTENTE (mail, username, passw)
		ON DELETE CASCADE,
	FOREIGN KEY (idPost)
		REFERENCES POST (id)
		ON DELETE CASCADE
);

CREATE TABLE FLAG_POST (
	MailU			VARCHAR (100)	NOT NULL,
	UsernameU		VARCHAR (30)	NOT NULL,
	PasswU			VARCHAR (300)	NOT NULL,
	IdPost			INT		NOT NULL,
	DataFlag		TIMESTAMP	NOT NULL,
	PRIMARY KEY (mailU, usernameU, passwU, idPost),
	FOREIGN KEY (mailU, usernameU, passwU)
		REFERENCES UTENTE (mail, username, passw)
		ON DELETE CASCADE,
	FOREIGN KEY (idPost)
		REFERENCES POST (id)
		ON DELETE CASCADE
);

CREATE TABLE CREA (
	MailU			VARCHAR (100)	NOT NULL,
	UsernameU		VARCHAR (30)	NOT NULL,
	PasswU			VARCHAR (300)	NOT NULL,
	IdPost			INT		NOT NULL,
	PRIMARY KEY (mailU, usernameU, passwU, idPost),
	FOREIGN KEY (mailU, usernameU, passwU)
		REFERENCES UTENTE (mail, username, passw)
		ON DELETE CASCADE,
	FOREIGN KEY (idPost)
		REFERENCES POST (id)
		ON DELETE CASCADE
);

CREATE TABLE LIKE_POST (
	MailU			VARCHAR (100)	NOT NULL,
	UsernameU		VARCHAR (30)	NOT NULL,
	PasswU			VARCHAR (300)	NOT NULL,
	IdPost			INT		NOT NULL,
	DataLike		TIMESTAMP	NOT NULL,
	PRIMARY KEY (mailU, usernameU, passwU, idPost),
	FOREIGN KEY (mailU, usernameU, passwU)
		REFERENCES UTENTE (mail, username, passw)
		ON DELETE CASCADE,
	FOREIGN KEY (idPost)
		REFERENCES POST (id)
		ON DELETE CASCADE
);

CREATE TABLE SEGUE (
	MailSeguito		VARCHAR (100)	NOT NULL,
	UsernameSeguito		VARCHAR (30)	NOT NULL,
	PasswSeguito		VARCHAR (300)	NOT NULL,
	MailSeguitore		VARCHAR (100)	NOT NULL,
	UsernameSeguitore	VARCHAR (30)	NOT NULL,
	PasswSeguitore		VARCHAR (300)	NOT NULL,
	DataFollow		TIMESTAMP	NOT NULL,
	PRIMARY KEY (mailSeguito, usernameSeguito, passwSeguito, mailSeguitore, usernameSeguitore, passwSeguitore),
	FOREIGN KEY (mailSeguito, usernameSeguito, passwSeguito)
		REFERENCES UTENTE (mail, username, passw)
		ON DELETE CASCADE,
	FOREIGN KEY (mailSeguitore, usernameSeguitore, passwSeguitore)
		REFERENCES UTENTE (mail, username, passw)
		ON DELETE CASCADE
);

-- Update contatori e ban utente

-- numFollow e numFollowers in UTENTE
UPDATE UTENTE u SET
  numFollow    = (SELECT COUNT(*) FROM SEGUE s WHERE s.MailSeguitore    = 		u.Mail AND s.UsernameSeguitore = u.Username AND 			s.PasswSeguitore = u.Passw),
  numFollowers = (SELECT COUNT(*) FROM SEGUE s WHERE s.MailSeguito     = 		u.Mail AND s.UsernameSeguito  = u.Username AND 				s.PasswSeguito    = u.Passw);

-- numLike e numFlag in POST
UPDATE POST p SET
  NumLike = (SELECT COUNT(*) FROM LIKE_POST  l WHERE l.IdPost = p.Id),
  NumFlag = (SELECT COUNT(*) FROM FLAG_POST  f WHERE f.IdPost = p.Id);

-- postModerati in UTENTE
UPDATE UTENTE u SET
  postModerati = (SELECT COUNT(*) FROM MODERA m WHERE m.MailU=u.Mail AND 		m.UsernameU=u.Username AND m.PasswU=u.Passw);

-- ban automatico in UTENTE
UPDATE UTENTE
SET èBannato = TRUE
WHERE postModerati > 3;
